cmake_minimum_required(VERSION 3.21)
project(qos_chromium LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT QOS_BLOCKER_LIBRARY OR NOT EXISTS "${QOS_BLOCKER_LIBRARY}")
  message(FATAL_ERROR "QOS_BLOCKER_LIBRARY must name the built Rust blocker static library")
endif()
if(NOT CEF_ROOT)
  message(FATAL_ERROR "CEF_ROOT must point to the pinned SDK; use scripts/browser/build-host.py")
endif()
if(NOT QOS_OUTPUT_DIR)
  get_filename_component(QOS_OUTPUT_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../target/chromium" ABSOLUTE)
endif()
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
if(APPLE)
  enable_language(OBJCXX)
endif()
if(CMAKE_SYSTEM_PROCESSOR MATCHES "^(arm64|aarch64)$")
  set(PROJECT_ARCH arm64)
else()
  set(PROJECT_ARCH x86_64)
endif()
set(USE_SANDBOX ON CACHE BOOL "Sandbox is mandatory" FORCE)
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" libcef_dll_wrapper)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${QOS_OUTPUT_DIR}")
function(qos_target target)
  target_include_directories(${target} PRIVATE "${CEF_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/src")
  target_compile_definitions(${target} PRIVATE ${CEF_COMPILER_DEFINES})
  target_link_libraries(${target} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS} "${QOS_BLOCKER_LIBRARY}")
endfunction()
if(APPLE)
  set(QOS_APP "${QOS_OUTPUT_DIR}/Browser Engine.app")
  add_executable(qos-chromium-host MACOSX_BUNDLE src/host.cc src/main_mac.mm)
  qos_target(qos-chromium-host)
  set_target_properties(qos-chromium-host PROPERTIES OUTPUT_NAME "Browser Engine" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/Info.plist")
  # The bundle name and executable name are intentionally different.
  add_custom_command(TARGET qos-chromium-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E rename "${QOS_APP}/Contents/MacOS/Browser Engine" "${QOS_APP}/Contents/MacOS/qos-chromium-host" VERBATIM)
  # Upstream's ln -sf follows existing directory symlinks on repeat builds,
  # creating broken self-links inside Versions/A. Recreate this generated copy
  # before invoking the upstream layout macro; the cached SDK is untouched.
  add_custom_command(TARGET qos-chromium-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E rm -rf "${QOS_APP}/Contents/Frameworks/Chromium Embedded Framework.framework" VERBATIM)
  COPY_MAC_FRAMEWORK(qos-chromium-host "${CEF_BINARY_DIR}" "${QOS_APP}")
  foreach(suffix_list ${CEF_HELPER_APP_SUFFIXES})
    string(REPLACE ":" ";" suffix "${suffix_list}")
    list(GET suffix 0 name_suffix)
    list(GET suffix 1 target_suffix)
    list(GET suffix 2 id_suffix)
    set(helper_target "qos-helper${target_suffix}")
    set(helper_name "Browser Engine Helper${name_suffix}")
    set(HELPER_EXECUTABLE "${helper_name}")
    set(HELPER_ID_SUFFIX "${id_suffix}")
    configure_file(resources/helper-Info.plist.in "${CMAKE_CURRENT_BINARY_DIR}/${helper_target}.plist" @ONLY)
    add_executable(${helper_target} MACOSX_BUNDLE src/helper_mac.cc src/renderer.cc)
    qos_target(${helper_target})
    set_target_properties(${helper_target} PROPERTIES OUTPUT_NAME "${helper_name}" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/${helper_target}.plist")
    add_dependencies(qos-chromium-host ${helper_target})
    add_custom_command(TARGET qos-chromium-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${QOS_OUTPUT_DIR}/${helper_name}.app" "${QOS_APP}/Contents/Frameworks/${helper_name}.app" VERBATIM)
  endforeach()
  add_custom_command(TARGET qos-chromium-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "${QOS_APP}/Contents/Frameworks/Browser Engine Helper.app/Contents/Resources" COMMAND codesign --force --deep --sign - "${QOS_APP}" VERBATIM)
elseif(UNIX)
  ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
  add_executable(qos-chromium-host src/host.cc src/renderer.cc src/main_linux.cc)
  qos_target(qos-chromium-host)
  target_link_libraries(qos-chromium-host PRIVATE libcef_lib)
  set_target_properties(qos-chromium-host PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  COPY_FILES(qos-chromium-host "${CEF_BINARY_FILES}" "${CEF_BINARY_DIR}" "${QOS_OUTPUT_DIR}")
  COPY_FILES(qos-chromium-host "${CEF_RESOURCE_FILES}" "${CEF_RESOURCE_DIR}" "${QOS_OUTPUT_DIR}")
else()
  message(FATAL_ERROR "Only macOS and Linux are supported")
endif()
enable_testing()
add_executable(qos-protocol-test tests/protocol_test.cc)
target_include_directories(qos-protocol-test PRIVATE src)
target_compile_options(qos-protocol-test PRIVATE -UNDEBUG)
add_test(NAME protocol COMMAND qos-protocol-test)
