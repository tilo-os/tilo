function(tilo_target target)
  target_include_directories(${target} PRIVATE "${CEF_ROOT}" "${CMAKE_CURRENT_SOURCE_DIR}/src" "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_compile_definitions(${target} PRIVATE ${CEF_COMPILER_DEFINES})
  target_link_libraries(${target} PRIVATE libcef_dll_wrapper ${CEF_STANDARD_LIBS} "${TILO_BLOCKER_LIBRARY}")
endfunction()
if(APPLE)
  set(TILO_APP "${TILO_OUTPUT_DIR}/Browser Engine.app")
  add_executable(tilo-chromium-host MACOSX_BUNDLE src/host.cc src/main_mac.mm)
  tilo_target(tilo-chromium-host)
  set_target_properties(tilo-chromium-host PROPERTIES OUTPUT_NAME "Browser Engine" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/resources/Info.plist")
  # The bundle name and executable name are intentionally different.
  add_custom_command(TARGET tilo-chromium-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E rename "${TILO_APP}/Contents/MacOS/Browser Engine" "${TILO_APP}/Contents/MacOS/tilo-chromium-host" VERBATIM)
  # Upstream's ln -sf follows existing directory symlinks on repeat builds,
  # creating broken self-links inside Versions/A. Recreate this generated copy
  # before invoking the upstream layout macro; the cached SDK is untouched.
  add_custom_command(TARGET tilo-chromium-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E rm -rf "${TILO_APP}/Contents/Frameworks/Chromium Embedded Framework.framework" VERBATIM)
  COPY_MAC_FRAMEWORK(tilo-chromium-host "${CEF_BINARY_DIR}" "${TILO_APP}")
  foreach(suffix_list ${CEF_HELPER_APP_SUFFIXES})
    string(REPLACE ":" ";" suffix "${suffix_list}")
    list(GET suffix 0 name_suffix)
    list(GET suffix 1 target_suffix)
    list(GET suffix 2 id_suffix)
    set(helper_target "tilo-helper${target_suffix}")
    set(helper_name "Browser Engine Helper${name_suffix}")
    set(HELPER_EXECUTABLE "${helper_name}")
    set(HELPER_ID_SUFFIX "${id_suffix}")
    configure_file(resources/helper-Info.plist.in "${CMAKE_CURRENT_BINARY_DIR}/${helper_target}.plist" @ONLY)
    add_executable(${helper_target} MACOSX_BUNDLE src/helper_mac.cc src/renderer.cc)
    tilo_target(${helper_target})
    set_target_properties(${helper_target} PROPERTIES OUTPUT_NAME "${helper_name}" MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_BINARY_DIR}/${helper_target}.plist")
    add_dependencies(tilo-chromium-host ${helper_target})
    add_custom_command(TARGET tilo-chromium-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory "${TILO_OUTPUT_DIR}/${helper_name}.app" "${TILO_APP}/Contents/Frameworks/${helper_name}.app" VERBATIM)
  endforeach()
  add_custom_command(TARGET tilo-chromium-host POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "${TILO_APP}/Contents/Frameworks/Browser Engine Helper.app/Contents/Resources" COMMAND codesign --force --deep --sign - "${TILO_APP}" VERBATIM)
elseif(UNIX)
  # The host renders offscreen and calls no X11 API; libcef.so carries its own
  # X11 dependency. Not linking it keeps libX11-devel out of the build.
  list(REMOVE_ITEM CEF_STANDARD_LIBS X11)
  ADD_LOGICAL_TARGET(libcef_lib "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
  add_executable(tilo-chromium-host src/host.cc src/renderer.cc src/main_linux.cc)
  tilo_target(tilo-chromium-host)
  target_link_libraries(tilo-chromium-host PRIVATE libcef_lib)
  # CEF's build defaults do not supply the distribution's executable hardening.
  # Set it on our host, independently of the SDK's prebuilt shared libraries.
  target_compile_options(tilo-chromium-host PRIVATE -fPIE -fstack-protector-strong -fstack-clash-protection -U_FORTIFY_SOURCE -D_FORTIFY_SOURCE=3)
  target_link_options(tilo-chromium-host PRIVATE -pie -Wl,-z,relro -Wl,-z,now -Wl,-z,noexecstack)
  if(PROJECT_ARCH STREQUAL "x86_64")
    target_compile_options(tilo-chromium-host PRIVATE -fcf-protection=full)
    target_link_options(tilo-chromium-host PRIVATE -Wl,-z,shstk)
  endif()
  # libcef.so's own dependencies (ATK, CUPS, NSS, X11 extensions) are resolved
  # where the engine runs; the image checks them there. The builder need not
  # carry them just to link the host.
  target_link_options(tilo-chromium-host PRIVATE -Wl,--allow-shlib-undefined)
  set_target_properties(tilo-chromium-host PROPERTIES BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
  COPY_FILES(tilo-chromium-host "${CEF_BINARY_FILES}" "${CEF_BINARY_DIR}" "${TILO_OUTPUT_DIR}")
  COPY_FILES(tilo-chromium-host "${CEF_RESOURCE_FILES}" "${CEF_RESOURCE_DIR}" "${TILO_OUTPUT_DIR}")
else()
  message(FATAL_ERROR "Only macOS and Linux are supported")
endif()
